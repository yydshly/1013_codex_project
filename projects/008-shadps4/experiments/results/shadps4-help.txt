shadPS4 Emulator CLI


<runtime>\shadps4\shadPS4.exe [OPTIONS] [guest_arg]


POSITIONALS:
  guest_arg TEXT              Game path or ID

OPTIONS:
  -h,     --help              Print this help message and exit
  -g,     --game TEXT         Game path or ID
  -p,     --patch TEXT        Patch file to apply
  -i,     --ignore-game-patch Disable automatic loading of game patches
  -b,     --big-picture       Start in Big Picture Mode
          --same-process      Launch the game in the same process when using Big Picture Mode
  -f,     --fullscreen TEXT   Fullscreen mode (true|false)
          --override-root TEXT:DIR 
          --wait-for-debugger 
          --wait-for-pid INT  
          --show-fps          
          --config-clean      
          --config-global     
          --log-append        
          --add-game-folder TEXT:DIR 
          --set-addon-folder TEXT:DIR 
          --mount [TEXT,TEXT] ... 
                              Mount source to destination
  -e,     --env TEXT ...      Environment variables to pass to the guest
